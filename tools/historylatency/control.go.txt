package ledger

import (
	"context"
	"encoding/json"
	"os"
	"path/filepath"
	"runtime"
	"slices"
	"strconv"
	"testing"
	"time"

	"github.com/agenxy/dibs/internal/core"
	"github.com/agenxy/dibs/internal/engine"
)

// A separate process supplies a no-warm baseline; no control ops or startup
// delay enter the original production process. The same paired ledger path
// is replaced by the original probe afterwards, avoiding an extra large copy.
func TestHistoryNoWarmAcceptanceControl(t *testing.T) {
	if os.Getenv("DIBS_HISTORY_REPEAT") == "" {
		t.Skip("hosted original-probe acceptance control")
	}
	count, err := strconv.Atoi(os.Getenv("DIBS_HISTORY_PROBE_RECORDS"))
	if err != nil || (count != 1_000_000 && count != 2_000_000) {
		t.Fatal("setup: invalid count", err)
	}
	dir := ".history-production-probe"
	box, err := LoadOrCreateKey(filepath.Join(dir, "key"))
	if err != nil {
		t.Fatal("setup:", err)
	}
	path := copyHistoryProbeLedger(t, filepath.Join(dir, "ledger.jsonl"), filepath.Join(dir, "paired.ledger.jsonl"))
	l, err := Open(path, "history-probe", box)
	if err != nil {
		t.Fatal("setup:", err)
	}
	defer func() { _ = l.Close() }()
	st := core.NewState("history-probe", core.DefaultLimits())
	n, err := l.Replay(st)
	if err != nil || n != count || st.Serial != uint64(count) {
		t.Fatal("setup: actual Replay", n, st.Serial, err)
	}
	if actor := st.Agents["party-00"]; actor == nil || actor.Token != "party-00" {
		t.Fatal("setup: actor missing")
	}
	ctx, cancel := context.WithTimeout(context.Background(), time.Minute)
	defer cancel()
	eng := engine.New(st, l, nil)
	joined := make(chan struct{})
	go func() { eng.Run(ctx); close(joined) }()
	defer func() { cancel(); <-joined }()
	if _, _, err := eng.SubscribeInfo(ctx, ""); err != nil {
		t.Fatal("setup:", err)
	}
	runtime.GC()
	tick := time.NewTicker(2 * time.Millisecond)
	defer tick.Stop()
	var samples []float64
	var mem runtime.MemStats
	for range 2048 {
		runtime.ReadMemStats(&mem) // same original sampler, before timed Do
		m := l.mail.Measurement()
		if m.Started || m.Ready || m.Failed {
			t.Fatal("setup: control warmed or failed", m)
		}
		if err := eng.SetRateTokens(ctx, "party-00", 30); err != nil {
			t.Fatal("setup:", err)
		}
		at := time.Now()
		res, err := eng.Do(ctx, &core.Op{Kind: core.OpAckBoard, Token: "party-00"})
		samples = append(samples, time.Since(at).Seconds())
		if err != nil || res["error"] != nil {
			t.Fatal("setup: actual no-warm writer op", err, res)
		}
		select {
		case <-tick.C:
		case <-ctx.Done():
			t.Fatal("setup:", ctx.Err())
		}
	}
	slices.Sort(samples)
	raw, err := json.Marshal(map[string]any{"samples": len(samples), "records": n, "p99_seconds": samples[(len(samples)-1)*99/100]})
	if err != nil {
		t.Fatal(err)
	}
	if err = os.WriteFile(filepath.Join(dir, "no-warm.json"), raw, 0o600); err != nil {
		t.Fatal(err)
	}
	t.Logf("HISTORY_ORIGINAL_NO_WARM %s", raw)
}
